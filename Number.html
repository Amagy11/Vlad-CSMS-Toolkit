// Обновленная функция рендера таблицы с исправленным поиском function renderTable() { const tbody = document.getElementById('tableBody'); const searchQuery = document.getElementById('search').value.trim().toLowerCase(); let filteredData = [...table1Data]; // Фильтры по статусу if (currentFilter === 'new') { filteredData = filteredData.filter(row => row.status === 'new'); } else if (currentFilter === 'inwork') { filteredData = filteredData.filter(row => row.status === 'inwork'); } else if (currentFilter === 'completed') { filteredData = filteredData.filter(row => row.status === 'completed'); } else if (currentFilter === 'highlighted') { filteredData = filteredData.filter(row => row.isBudget || row.isUrgent || row.isDebt); } else if (currentFilter === 'matched') { filteredData = filteredData.filter(row => row.isMatched); } // Фильтр по типу прибора if (selectedDevice) { filteredData = filteredData.filter(row => { const device = row['Прибор'] || ''; return device.split(' ').slice(0, 3).join(' ') === selectedDevice; }); } // Поиск (улучшенный) if (searchQuery) { filteredData = filteredData.filter(row => { const invoiceNumber = (row['Номер счета'] || '').toString().toLowerCase(); const contractor = (row['Контрагент (Наименование)'] || '').toLowerCase(); const device = (row['Прибор'] || '').toLowerCase(); const paymentDate = (row.paymentDate || row['Дата оплаты'] || '').toLowerCase(); const urgency = (row.urgency || '').toLowerCase(); // Поиск по номеру счета (можно искать часть номера) if (invoiceNumber.includes(searchQuery)) return true; // Поиск по контрагенту if (contractor.includes(searchQuery)) return true; // Поиск по прибору if (device.includes(searchQuery)) return true; // Поиск по дате оплаты if (paymentDate.includes(searchQuery)) return true; // Поиск по срочности if (urgency.includes(searchQuery)) return true; return false; }); } // Сортировка по дате оплаты if (sortByDate === 'earliest') { filteredData.sort((a, b) => { const dateA = parseDate(a.paymentDate || a['Дата оплаты']); const dateB = parseDate(b.paymentDate || b['Дата оплаты']); if (!dateA && !dateB) return 0; if (!dateA) return 1; if (!dateB) return -1; return dateA - dateB; }); } else if (sortByDate === 'latest') { filteredData.sort((a, b) => { const dateA = parseDate(a.paymentDate || a['Дата оплаты']); const dateB = parseDate(b.paymentDate || b['Дата оплаты']); if (!dateA && !dateB) return 0; if (!dateA) return 1; if (!dateB) return -1; return dateB - dateA; }); } else if (sortByDate === 'noDate') { filteredData = filteredData.filter(row => !row.paymentDate && !row['Дата оплаты'] ); } // Обновление индикатора сортировки const dateSortIndicator = document.getElementById('dateSortIndicator'); if (sortByDate === 'earliest') { dateSortIndicator.textContent = '↑'; } else if (sortByDate === 'latest') { dateSortIndicator.textContent = '↓'; } else if (sortByDate === 'noDate') { dateSortIndicator.textContent = '∅'; } else { dateSortIndicator.textContent = ''; } // Статистика document.getElementById('totalCount').textContent = table1Data.length; document.getElementById('newCount').textContent = table1Data.filter(r => r.status === 'new').length; document.getElementById('inworkCount').textContent = table1Data.filter(r => r.status === 'inwork').length; document.getElementById('completedCount').textContent = table1Data.filter(r => r.status === 'completed').length; document.getElementById('highlightedCount').textContent = table1Data.filter(r => r.isBudget || r.isUrgent || r.isDebt).length; document.getElementById('urgentCount').textContent = table1Data.filter(r => r.isUrgent).length; // Заголовок const titles = { 'all': 'Все счета', 'new': 'Новые счета', 'inwork': 'Счета в работе', 'completed': 'Выполненные счета', 'highlighted': 'Счета с пометками', 'matched': 'Совпавшие счета' }; let title = titles[currentFilter] || 'Счета'; if (sortByDate === 'earliest') title += ' (сначала ранние)'; if (sortByDate === 'latest') title += ' (сначала поздние)'; if (sortByDate === 'noDate') title += ' (без даты оплаты)'; if (searchQuery) title += ` (поиск: "${searchQuery}")`; document.getElementById('tableTitle').textContent = `📋 ${title} (${filteredData.length})`; // Рендер tbody.innerHTML = ''; if (filteredData.length === 0) { tbody.innerHTML = 'Нет данных. Измените фильтры или поиск.'; return; } filteredData.forEach(row => { const tr = document.createElement('tr'); if (row.status === 'inwork') tr.classList.add('status-inwork'); if (row.status === 'completed') tr.classList.add('status-completed'); if (row.isBudget) tr.classList.add('budget'); if (row.isUrgent) tr.classList.add('urgent'); if (row.isDebt) tr.classList.add('debt'); if (row.isBudget || row.isUrgent || row.isDebt) tr.classList.add('highlighted'); const paymentDate = row.paymentDate || row['Дата оплаты'] || ''; tr.innerHTML = ` ${row['Номер счета'] || ''} ${row['Дата выписки'] || ''} ${row['Контрагент (Наименование)'] || ''} ${row['Прибор'] || ''} ${row['Кол-во'] || ''} ${row['Всего по счету'] || row['Сумма'] || ''} ${row.paymentStatus ? `${row.paymentStatus}` : '—'} ${paymentDate ? `${paymentDate}` : '—'} ${row.urgency ? `${row.urgency} дн.` : '—'} ${row.status === 'new' ? 'Новый' : ''} ${row.status === 'inwork' ? 'В работе' : ''} ${row.status === 'completed' ? '✓ Выполнен' : ''} ${row.isBudget ? 'Бюджет' : ''} ${row.isUrgent ? 'Срочно' : ''} ${row.isDebt ? 'Задолж.' : ''} ${!row.isBudget && !row.isUrgent && !row.isDebt ? '—' : ''} ${row.status === 'new' ? ` ` : ''} ${row.status === 'inwork' ? ` ` : ''} ${row.status === 'completed' ? ` ` : ''} `; tbody.appendChild(tr); }); }
