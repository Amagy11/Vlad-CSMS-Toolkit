function readTable1(file, callback) { const reader = new FileReader(); reader.onload = function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, {type: 'array'}); let allData = []; workbook.SheetNames.forEach(sheetName => { const sheet = workbook.Sheets[sheetName]; const jsonData = XLSX.utils.sheet_to_json(sheet, { defval: '', raw: true, header: 1 }); console.log(`--- Лист "${sheetName}": строк ${jsonData.length} ---`); let headers = null; let section = 'common'; let lastInvoiceNumber = ''; let lastInvoiceDate = ''; let lastContractor = ''; for (let i = 0; i < jsonData.length; i++) { const row = jsonData[i]; // Пустая строка — сброс раздела и протяжки if (isEmptyRow(row)) { section = 'common'; lastInvoiceNumber = ''; lastInvoiceDate = ''; lastContractor = ''; continue; } // Маркер раздела const marker = detectSectionMarker(row); if (marker) { section = marker; headers = null; // заголовки ищем заново lastInvoiceNumber = ''; lastInvoiceDate = ''; lastContractor = ''; continue; } // Заголовок таблицы? if (isHeaderRow(row)) { headers = row.map(h => h ? h.toString().trim() : ''); lastInvoiceNumber = ''; lastInvoiceDate = ''; lastContractor = ''; console.log(`Заголовки на строке ${i}:`, headers); continue; } // Если заголовков ещё нет — пропускаем строку if (!headers) { console.log(`Строка ${i}: нет заголовков, пропуск`); continue; } // Разбираем строку данных let invoiceNumber = '', invoiceDate = '', contractor = ''; let device = '', quantity = '', amount = ''; let totalAmount = '', paymentDate = ''; headers.forEach((header, index) => { const value = row[index] || ''; const headerLower = header.toLowerCase(); if (headerLower.includes('номер') && headerLower.includes('счет')) { invoiceNumber = value.toString().trim(); } else if (headerLower.includes('номер') && headerLower.includes('квитанц')) { invoiceNumber = value.toString().trim(); } else if (headerLower.includes('дата') && headerLower.includes('выписк')) { invoiceDate = value.toString().trim(); } else if (headerLower.includes('контрагент') || headerLower.includes('наименование')) { contractor = value.toString().trim(); } else if (headerLower.includes('прибор')) { device = value.toString().trim(); } else if (headerLower.includes('кол-во') || headerLower.includes('колич')) { quantity = value.toString().trim(); } else if (headerLower.includes('сумм') || headerLower.includes('всего')) { amount = value.toString().trim(); totalAmount = value.toString().trim(); } else if (headerLower.includes('оплат')) { paymentDate = value.toString().trim(); } }); // Протягиваем номер счёта вниз (объединённые ячейки) if (!invoiceNumber && lastInvoiceNumber) { // Строка без номера — считаем продолжением предыдущего счёта // Только если есть хоть какие-то данные в строке const hasAnyData = device || quantity || amount; if (hasAnyData) { invoiceNumber = lastInvoiceNumber; if (!invoiceDate) invoiceDate = lastInvoiceDate; if (!contractor) contractor = lastContractor; } } // Запоминаем последние непустые значения if (invoiceNumber) lastInvoiceNumber = invoiceNumber; if (invoiceDate) lastInvoiceDate = invoiceDate; if (contractor) lastContractor = contractor; // Проверка: есть ли номер счёта с дефисом if (!invoiceNumber || !invoiceNumber.includes('-')) { // Не мусорим в консоль, если строка совсем пустая if (device || quantity || amount || contractor) { console.log(`Строка ${i}: пропущена (нет номера счёта с дефисом). Значения:`, {invoiceNumber, device, quantity, amount, contractor}); } continue; } // Добавляем строку allData.push({ 'Номер счета': invoiceNumber, 'Дата выписки': invoiceDate, 'Контрагент (Наименование)': contractor, 'Прибор': device, 'Кол-во': quantity, 'Сумма': amount, 'Всего по счету': totalAmount || amount, 'Оплата': '', 'Дата оплаты': paymentDate, section: section, isBudget: section === 'budget', isUrgent: section === 'urgent', isDebt: section === 'debt', isMatched: false, status: 'new', statusDate: null, paymentStatus: '', urgency: '' }); } }); console.log(`=== Таблица 1: загружено ${allData.length} позиций ===`); callback(allData); } catch (error) { console.error('Ошибка:', error); alert('Ошибка чтения файла: ' + error.message); } }; reader.readAsArrayBuffer(file); }
