// Исправленная функция чтения Таблицы 2 function readTable2(file, callback) { const reader = new FileReader(); reader.onload = function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, {type: 'array'}); let allData = []; workbook.SheetNames.forEach(sheetName => { const sheet = workbook.Sheets[sheetName]; const jsonData = XLSX.utils.sheet_to_json(sheet, { defval: '', raw: true, header: 1 }); console.log(`=== Лист "${sheetName}" (Таблица 2) ===`); console.log(`Всего строк: ${jsonData.length}`); // Выводим первые 5 строк для отладки for (let i = 0; i < Math.min(5, jsonData.length); i++) { console.log(`Строка ${i + 1}:`, jsonData[i]); } let headerRowIndex = -1; let headers = []; // Ищем шапку - более гибкий поиск for (let i = 0; i < jsonData.length; i++) { const row = jsonData[i]; // Проверяем, есть ли в строке что-то похожее на номер счета const hasNumber = row.some(cell => cell && cell.toString().includes('-') && /\d/.test(cell.toString()) ); // Ищем заголовки const hasHeader = row.some(cell => cell && ( cell.toString().toLowerCase().includes('номер') || cell.toString().toLowerCase().includes('квитанц') || cell.toString().toLowerCase().includes('счет') || cell.toString().toLowerCase().includes('бюджет') || cell.toString().toLowerCase().includes('срочн') ) ); if (hasHeader && !hasNumber) { headerRowIndex = i; headers = row.map(h => h ? h.toString().trim() : ''); console.log(`✅ Шапка найдена в строке ${i + 1}:`, headers); break; } } if (headerRowIndex === -1) { console.log('❌ Шапка не найдена'); return; } // Обрабатываем данные for (let i = headerRowIndex + 1; i < jsonData.length; i++) { const row = jsonData[i]; let invoiceNumber = ''; let paymentDate = ''; let isBudget = false; let urgency = ''; let paymentStatus = ''; let contractor = ''; let amount = ''; // Перебираем все ячейки row.forEach((value, index) => { const header = headers[index] || ''; const headerLower = header.toLowerCase(); const cellValue = value ? value.toString().trim() : ''; // Номер счета/квитанции if (headerLower.includes('квитанц') || headerLower.includes('номер') || headerLower.includes('счет')) { if (cellValue && cellValue.includes('-')) { invoiceNumber = cellValue; } } // Дата оплаты if (headerLower.includes('дата') && headerLower.includes('оплат')) { paymentDate = cellValue; } // Бюджет if (headerLower.includes('бюджет')) { console.log(`Бюджет: header="${header}", value="${cellValue}"`); if (cellValue.toLowerCase() === 'да' || cellValue.toLowerCase() === 'yes' || cellValue === '1' || cellValue === '+') { isBudget = true; } } // Срочность if (headerLower.includes('срочн')) { console.log(`Срочность: header="${header}", value="${cellValue}"`); urgency = cellValue; } // Наличие оплаты if (headerLower.includes('наличие') && headerLower.includes('оплат')) { paymentStatus = cellValue; } // Контрагент if (headerLower.includes('контрагент') || headerLower.includes('наименование')) { contractor = cellValue; } // Сумма if (headerLower.includes('сумм')) { amount = cellValue; } }); // Если нашли номер счета if (invoiceNumber && invoiceNumber.includes('-')) { const rowData = { 'Номер счета': invoiceNumber, 'Дата оплаты': paymentDate, 'Б': isBudget ? 'Да' : '', 'С': urgency, 'З': '', 'Наличие оплаты': paymentStatus, 'Контрагент': contractor, 'Сумма': amount, '_raw': { 'isBudget': isBudget, 'urgency': urgency } }; allData.push(rowData); console.log(`✅ Найден счет: ${invoiceNumber}, бюджет: ${isBudget}, срочность: ${urgency}`); } } }); // Удаляем дубликаты const uniqueData = []; const seen = new Set(); allData.forEach(row => { if (!seen.has(row['Номер счета'])) { seen.add(row['Номер счета']); uniqueData.push(row); } }); console.log(`=== Итого найдено в Таблице 2: ${uniqueData.length} записей ===`); console.log('Примеры с пометками:', uniqueData.filter(r => r['Б'] === 'Да' || r['С']).slice(0, 5)); callback(uniqueData); } catch (error) { console.error('Ошибка чтения Таблицы 2:', error); alert('Ошибка чтения Таблицы 2: ' + error.message); } }; reader.readAsArrayBuffer(file); } // Исправленная функция сопоставления function matchTables() { if (!table1Data.length || !table2Data.length) return; console.log('=== Сопоставление таблиц ==='); console.log('Таблица 1:', table1Data.length, 'записей'); console.log('Таблица 2:', table2Data.length, 'записей'); const table2Map = {}; table2Data.forEach(row => { const num = row['Номер счета']?.trim(); if (num) { table2Map[num] = row; } }); console.log('Номера счетов в Таблице 2:', Object.keys(table2Map).slice(0, 10)); let matchCount = 0; table1Data = table1Data.map(row => { const num = row['Номер счета']?.trim(); const matched = table2Map[num]; if (matched) { matchCount++; // Проверяем пометки const isBudget = matched['Б'] === 'Да' || matched['Б'] === 'да' || matched['Б'] === true; const isUrgent = matched['С'] && matched['С'].trim() !== ''; console.log(`Совпадение: ${num}, бюджет: ${isBudget}, срочность: ${matched['С']}`); return { ...row, isMatched: true, isBudget: isBudget, isUrgent: isUrgent, urgency: matched['С'] || '', paymentStatus: matched['Наличие оплаты'] || '', paymentDate: matched['Дата оплаты'] || row['Дата оплаты'] || '' };
