// ============================================================ // ⭐ ОСНОВНОЕ СРАВНЕНИЕ (С ИНДЕКСАЦИЕЙ) // ============================================================ document.getElementById('compareBtn').addEventListener('click', async function() { if (!leftData || !rightData) return; const mainKeys = ['type', 'registry', 'sut']; const missingLeft = mainKeys.filter(k => !leftColMap[k]); const missingRight = mainKeys.filter(k => !rightColMap[k]); if (missingLeft.length || missingRight.length) { alert('Не найдены все основные колонки! Проверьте названия.'); return; } document.getElementById('loadingOverlay').classList.add('active'); document.getElementById('loadingText').textContent = `Сравнение ${leftData.length} строк...`; await new Promise(r => setTimeout(r, 80)); const lName = leftColMap['type']; const lReg = leftColMap['registry']; const lSut = leftColMap['sut']; const lDoc = leftColMap['document'] || null; const lSticker = leftColMap['sticker'] || null; const rName = rightColMap['type']; const rReg = rightColMap['registry']; const rSut = rightColMap['sut']; const rDoc = rightColMap['document'] || null; const rSticker = rightColMap['sticker'] || null; const fieldNames = ['тип СИ', 'номер ГРСИ', 'номер СУТ']; const results = []; let okCount = 0, errorCount = 0; // ============================================================ // ШАГ 1: ПОСТРОЕНИЕ ИНДЕКСА (ОДИН РАЗ) // ============================================================ console.log('⏳ Построение индекса справочника...'); const index = new Map(); for (let idx = 0; idx < rightData.length; idx++) { const row = rightData[idx]; const key = [ normalize(row[rName] || ''), normalize(row[rReg] || ''), normalize(row[rSut] || '') ].join('|||'); if (!index.has(key)) { index.set(key, []); } index.get(key).push(idx); } console.log(`✅ Индекс построен: ${index.size} уникальных ключей`); // ============================================================ // ШАГ 2: БЫСТРЫЙ ПОИСК ПО ИНДЕКСУ // ============================================================ for (let i = 0; i < leftData.length; i++) { const leftRow = leftData[i]; const leftKey = [ normalize(leftRow[lName] || ''), normalize(leftRow[lReg] || ''), normalize(leftRow[lSut] || '') ].join('|||'); const leftDisplay = [ leftRow[lName] || '', leftRow[lReg] || '', leftRow[lSut] || '' ]; const docValue = lDoc ? (leftRow[lDoc] || '') : ''; const stickerValue = lSticker ? (leftRow[lSticker] || '') : ''; // Ищем в индексе const matches = index.get(leftKey); if (matches && matches.length > 0) { // ✅ Найдено полное совпадение okCount++; const rightRow = rightData[matches[0]]; const rightDisplay = [ rightRow[rName] || '', rightRow[rReg] || '', rightRow[rSut] || '' ]; const rDocValue = rDoc ? (rightRow[rDoc] || '') : ''; const rStickerValue = rSticker ? (rightRow[rSticker] || '') : ''; results.push({ rowNum: i + 1, leftDisplay: leftDisplay, rightDisplay: rightDisplay, doc: docValue || rDocValue, sticker: stickerValue || rStickerValue, status: '✅ Совпадает', reason: '' }); } else { // ❌ Не найдено errorCount++; results.push({ rowNum: i + 1, leftDisplay: leftDisplay, rightDisplay: ['', '', ''], doc: docValue, sticker: stickerValue, status: '❌ Не найдено', reason: 'Нет совпадения в справочнике' }); } } document.getElementById('loadingOverlay').classList.remove('active'); allResults = results; showResults(results, okCount, errorCount); applyFilter('errors'); });
